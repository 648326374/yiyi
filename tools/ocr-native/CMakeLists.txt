cmake_minimum_required(VERSION 3.24)
project(YiyiOcr LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
if(NOT MSVC OR NOT RAPID_SOURCE OR NOT ORT_ROOT)
    message(FATAL_ERROR "Build with MSVC x64 and explicit RAPID_SOURCE / ORT_ROOT. See scripts/setup-ocr.ps1")
endif()
set(OpenCV_DIR "${RAPID_SOURCE}/opencv-static/windows-x64/x64/vc16/staticlib")
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/patched")
file(GLOB headers "${RAPID_SOURCE}/include/*.h")
foreach(header IN LISTS headers)
    get_filename_component(name "${header}" NAME)
    file(READ "${header}" content)
    string(REPLACE "Ort::Session *session;" "Ort::Session *session = nullptr;" content "${content}")
    string(REPLACE "<onnxruntime/core/session/onnxruntime_cxx_api.h>" "<onnxruntime_cxx_api.h>" content "${content}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/patched/${name}" "${content}")
endforeach()
set(sources)
foreach(name AngleNet CrnnNet DbNet OcrLite OcrUtils clipper)
    file(READ "${RAPID_SOURCE}/src/${name}.cpp" content)
    string(REPLACE "//sessionOptions.SetIntraOpNumThreads(numThread);" "sessionOptions.SetIntraOpNumThreads(numThread);\n    sessionOptions.AddConfigEntry(\"session.intra_op.allow_spinning\", \"0\");" content "${content}")
    string(REPLACE "sessionOptions.SetInterOpNumThreads(numThread);" "sessionOptions.SetInterOpNumThreads(1);" content "${content}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/patched/${name}.cpp" "${content}")
    list(APPEND sources "${CMAKE_CURRENT_BINARY_DIR}/patched/${name}.cpp")
endforeach()
add_library(YiyiOcr SHARED bridge.cpp ${sources})
target_compile_definitions(YiyiOcr PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS UNICODE _UNICODE)
target_compile_options(YiyiOcr PRIVATE /utf-8)
target_include_directories(YiyiOcr PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/patched" "${RAPID_SOURCE}/include" "${ORT_ROOT}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(YiyiOcr PRIVATE "${ORT_ROOT}/lib/onnxruntime.lib" ${OpenCV_LIBS})
